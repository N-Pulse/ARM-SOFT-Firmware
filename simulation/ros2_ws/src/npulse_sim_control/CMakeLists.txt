cmake_minimum_required(VERSION 3.16)
project(npulse_sim_control)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(controller_interface REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/ActuatorCommand.msg" "msg/ActuatorState.msg" "msg/FirmwareCommand.msg" "msg/FirmwareState.msg" "msg/HandCommand.msg" "msg/HandState.msg" DEPENDENCIES builtin_interfaces)
add_library(npulse_actuator_controller SHARED src/actuator_controller.cpp)
ament_target_dependencies(npulse_actuator_controller controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle)
rosidl_get_typesupport_target(typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(npulse_actuator_controller "${typesupport_target}")
pluginlib_export_plugin_description_file(controller_interface controller.xml)
install(TARGETS npulse_actuator_controller LIBRARY DESTINATION lib)
ament_export_dependencies(rosidl_default_runtime)
include("${CMAKE_CURRENT_SOURCE_DIR}/lockstep.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/wholehand.cmake")
ament_package()
